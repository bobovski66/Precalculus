% MODULE 8 PREQUIZ
\assessmenttitle{8}{Implicit Curves and Conics -- Prequiz}
\studentline
\directions{Four short questions. Read each equation geometrically.}
\begin{enumerate}
\item Identify the conic: $\dfrac{x^2}{16}+\dfrac{y^2}{9}=1$.
\answerblank{0.40in}
\item For $(x-2)^2+(y+1)^2=25$, state the center and radius.
\answerblank{0.48in}
\item For $x^2=-12y$, state the direction in which the parabola opens.
\answerblank{0.42in}
\item For $\dfrac{x^2}{9}-\dfrac{y^2}{4}=1$, state the vertices.
\answerblank{0.58in}
\end{enumerate}
